\begin{section}{Real Numbers and Limits of Sequences}

\begin{subsection}{Problem 1}
    \begin{frame}
        \begin{block}{Problem 1}
            Suppose that $\{a_n\}_{n=1}^\infty$ is a sequence of real numbers converging to a real number $L$. Determine the value of the following limit:
            \[
            \lim_{n \to \infty} \left( \frac{a_1 + 2a_2 + \cdots + n a_n}{1 + 2 + \cdots + n} \right)
            \]
            You are required to prove your conclusion using $\epsilon$-$N$ language.
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
            Let $b_n = a_n - L$. Since $b_n \to 0$, it suffices to prove the case where the limit is $0$.
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 2}
            Decompose the numerator into two parts:
            \[
                \sum_{k=1}^n k b_k = \underbrace{\sum_{k=1}^{N} k b_k}_{\text{Fixed finite sum}} + \underbrace{\sum_{k=N+1}^n k b_k}_{\text{Controlled by } \epsilon}
            \]
        \end{block}

    \end{frame}
\end{subsection}

\begin{subsection}{Problem 2}
    \begin{frame}
        \begin{block}{Problem 2}
            Let $c \in (0, \frac{3}{4})$ and $x_1 \in (0, 1)$. Define the sequence $\{x_n\}_{n=1}^{\infty}$ by the recurrence relation $x_{n+1} = 1 - c x_n^2$. Given that all terms of $\{x_n\}_{n=1}^{\infty}$ are distinct, prove that the sequence $\{x_n\}$ converges and find its limit.    
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
            Let $f(x) = 1 - cx^2$. Note that $f(x)$ is decreasing on $(0, 1)$.
            Therefore, the sequence $\{x_n\}$ is not monotonic.
            Consider the monotonicity of the composite function $g(x) = f(f(x))$.
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 2}
            Investigate the odd terms $\{x_{2n-1}\}$ and even terms $\{x_{2n}\}$ separately.
            Since $g(x)$ is increasing, these subsequences are monotonic.
            Show that they are bounded and thus convergent.
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 3}
            Let $x_{2n} \to \alpha$ and $x_{2n-1} \to \beta$. We have:
            \[
                \begin{cases}
                    \beta = 1 - c\alpha^2 \\
                    \alpha = 1 - c\beta^2
                \end{cases}
            \]
            Subtract the two equations: $\alpha - \beta = c(\alpha^2 - \beta^2)$.
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 4}
            Factor the difference to get $(\alpha - \beta)[1 - c(\alpha + \beta)] = 0$.
            Assume $\alpha \neq \beta$. Derive a contradiction using the given condition $c < 3/4$.
        \end{block}

    \end{frame}
\end{subsection}

\begin{subsection}{Problem 3}
    \begin{frame}
        \begin{block}{Problem 3}
            Given that $\lim\limits_{n \to \infty} (2x_{n+1} + x_n) = A \in \mathbb{R}$. Prove that the sequence $\{x_n\}$ converges.    
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
            Let $y_n = 2x_{n+1} + x_n$. We are given that $y_n \to A$.
            Rewrite the recurrence relation to express $x_{n+1}$ in terms of $x_n$:
            \[
                x_{n+1} = -\frac{1}{2}x_n + \frac{1}{2}y_n
            \]
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 2}
            Assume $\lim\limits_{n \to \infty} x_n = L$ exists. Taking the limit on both sides of the equation from Hint 1:
            \[
                L = -\frac{1}{2}L + \frac{1}{2}A
            \]
            Solve for $L$.
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 3}
            Let $L = A/3$. Consider the distance $|x_{n+1} - L|$.
            Use the Triangle Inequality to show:
            \[
                |x_{n+1} - L| \le \frac{1}{2}|x_n - L| + \frac{1}{2}|y_n - A|
            \]
            Since $\frac{1}{2} < 1$ and $|y_n - A| \to 0$, deduce that $|x_n - L| \to 0$.
        \end{block}
    \end{frame}

\end{subsection}

\begin{subsection}{Problem 4}
    \begin{frame}
        \begin{block}{Problem 4}
            Calculate the limit of the sequence:
            \[
            \lim_{n \to +\infty} \left( \frac{\ln 1 + 2\ln 2 + 3\ln 3 + \cdots + n\ln n}{n^2} - \frac{\ln n}{2} \right).
            \]    
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
            Let $S_n = \sum\limits_{k=1}^n k \ln k$.
            First, combine the two terms into one fraction to apply the \textbf{Stolz Theorem}:
            \[
                \lim_{n \to +\infty} \frac{S_n - \frac{1}{2}n^2 \ln n}{n^2}
            \]
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 2}
            When calculating the difference of the numerator, you will encounter the term $(n-1)^2 \ln(n-1)$.
            Rewrite it using logarithms:
            \[
                \ln(n-1) = \ln\left[n\left(1 - \frac{1}{n}\right)\right] = \ln n + \ln\left(1 - \frac{1}{n}\right)
            \]
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 3}
        To evaluate the limit, use the second-order Taylor expansion for $\ln(1-x)$ as $x \to 0$:
        \[
            \ln\left(1 - \frac{1}{n}\right) = -\frac{1}{n} - \frac{1}{2n^2} + o\left(\frac{1}{n^2}\right)
        \]
        Substitute this back to simplify the numerator difference.
        \end{block}
    \end{frame}

\end{subsection}

\end{section}

\begin{section}{Continuity of Functions}

\begin{subsection}{Problem 1}
    \begin{frame}
        \begin{block}{Problem 1}
            Let $a_1, a_2, \dots, a_n$ be arbitrary positive numbers. Calculate the limit:
            \[
                l = \lim_{x \to 0} \left( \frac{a_1^x + a_2^x + \cdots + a_n^x}{n} \right)^{\frac{1}{x}}.
            \]
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
        Recall the standard result: if $\lim\limits_{x \to 0} \alpha(x) = 0$, then
        \[
            \lim_{x \to 0} [1 + \alpha(x)]^{\frac{1}{x}} = e^{\lim\limits_{x \to 0} \frac{\alpha(x)}{x}}
        \]
        In this problem, let the base be $1 + \alpha(x)$, so:
        \[
            \alpha(x) = \frac{a_1^x + \cdots + a_n^x}{n} - 1 = \frac{\sum_{i=1}^n (a_i^x - 1)}{n}
        \]
        \end{block}

    \end{frame} 
    \begin{frame}
        \begin{block}{Hint 2}
        To evaluate the exponent limit $\lambda = \lim_{x \to 0} \frac{\alpha(x)}{x}$, rewrite it as:
        \[
            \lim_{x \to 0} \frac{1}{n} \sum_{i=1}^n \frac{a_i^x - 1}{x}
        \]
        Use the standard limit (definition of derivative at $x=0$):
        \[
            \lim_{x \to 0} \frac{a^x - 1}{x} = \ln a
        \]
        The final answer involves the sum of logarithms.
        \end{block}
    \end{frame}

\end{subsection}

\end{section}

\begin{section}{Derivatives and Differentials of Functions}

\begin{subsection}{Problem 1}
    \begin{frame}
        \begin{block}{Problem 1}
            Calculate the following limit:
            \[
            \lim_{x \to 0} \frac{1 - \cos x \sqrt{\cos 2x} \sqrt[3]{\cos 3x}}{x^2}.
            \]
        \end{block}
    \end{frame}

    \begin{frame}
        \begin{block}{Hint 1}
            Recall the second-order approximations:
            \[
                \cos u \approx 1 - \frac{1}{2}u^2
            \]
            \[
                (1 + u)^\alpha \approx 1 + \alpha u
            \]
        \end{block}

        \pause
        \vspace{2em}
        \begin{block}{Hint 2}
            Apply the approximations to the composite functions.
            \[
                \sqrt{\cos 2x} = (\cos 2x)^{\frac{1}{2}} \approx \left(1 - \frac{1}{2}(2x)^2\right)^{\frac{1}{2}} \approx 1 - x^2
            \]
            Do the same for $\sqrt[3]{\cos 3x}$, then multiply the linear approximations.
        \end{block}
    \end{frame}

\end{subsection}

\end{section}
